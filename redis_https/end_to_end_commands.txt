# End-to-end one-line commands (Maze/Mission JSON -> HTTPS -> AI Server -> Redis)

# 1) Go to redis_https folder on AI server
cd ~/abcapsp26TuThT3/redis_https

# 2) Build server binary
PKG_CONFIG_PATH=$HOME/local/lib/pkgconfig gcc -O2 -Wall -Wextra -std=c11 maze_https_mongo.c -o maze_https_redis $(pkg-config --cflags --libs libmicrohttpd gnutls libbson-1.0) -I$HOME/local/include -L$HOME/local/lib -lhiredis -Wl,-rpath,$HOME/local/lib

# 3) Check Redis is reachable
redis-cli -h 127.0.0.1 -p 6379 PING

# 4) Start HTTPS Redis server (keep this running terminal)
REDIS_HOST=127.0.0.1 REDIS_PORT=6379 REDIS_KEY_PREFIX=team3ttmission ./maze_https_redis

# 5) Health check from another terminal
curl --cacert certs/ca.crt --cert certs/client.crt --key certs/client.key https://localhost:8445/health

# 6) Send mission JSON over HTTPS (server -> Redis write)
curl --cacert certs/ca.crt --cert certs/client.crt --key certs/client.key -X POST https://localhost:8445/mission -H "Content-Type: application/json" -d '{"mission_id":"TEST_MISSION","robot_id":"TEST_ROBOT","mission_type":"patrol","start_time":1770056813,"end_time":1770056848,"moves_left_turn":46,"moves_right_turn":46,"moves_straight":52,"moves_reverse":8,"moves_total":152,"distance_traveled":24.41,"duration_seconds":35,"mission_result":"success","abort_reason":"user exited"}'

# 7) Read mission back via HTTPS GET endpoint
curl --cacert certs/ca.crt --cert certs/client.crt --key certs/client.key "https://localhost:8445/mission?mission_id=TEST_MISSION"

# 8) Verify mission directly in Redis
redis-cli HGETALL team3ttmission:TEST_MISSION

# 9) Maze app path: send directly from maze client (run where maze binary exists)
TELEMETRY_URL="https://10.170.8.109:8445/mission" CLIENT_CERT="redis_https/certs/client.crt" CLIENT_KEY="redis_https/certs/client.key" CA_CERT="redis_https/certs/ca.crt" ./maze_sdl2
