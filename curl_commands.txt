# curl Commands
#
# Run these from the repository root (so cert paths resolve).
#

## HTTP
# List telemetry (GET /moves)
curl -sS "http://localhost:8080/moves?limit=1000&sort=desc"

# Send telemetry (POST /move)
curl -sS -X POST "http://localhost:8080/move" \
  -H "Content-Type: application/json" \
  -d "{\"event_type\":\"player_move\",\"input\":{\"device\":\"keyboard\",\"move_sequence\":1},\"player\":{\"position\":{\"x\":1,\"y\":0}},\"goal_reached\":false,\"timestamp\":\"2026-02-13T00:00:00Z\"}"

## HTTPS (mTLS)
# List telemetry (GET /moves)
curl -sS --cacert https/certs/ca.crt --cert https/certs/client.crt --key https/certs/client.key \
  "https://localhost:8445/moves?limit=1000&sort=desc"

# Send telemetry (POST /move)
curl -sS --cacert https/certs/ca.crt --cert https/certs/client.crt --key https/certs/client.key \
  -X POST "https://localhost:8445/move" \
  -H "Content-Type: application/json" \
  -d "{\"event_type\":\"player_move\",\"input\":{\"device\":\"keyboard\",\"move_sequence\":1},\"player\":{\"position\":{\"x\":1,\"y\":0}},\"goal_reached\":false,\"timestamp\":\"2026-02-13T00:00:00Z\"}"
