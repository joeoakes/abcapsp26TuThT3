# mTLS + Redis + MongoDB — End-to-End Demo
# Run all commands from ~/abcapsp26TuThT3/https_final on the server
# Server: 10.170.8.109 (AI server)

# ─────────────────────────────────────────
# STEP 1 — Build the binary
# ─────────────────────────────────────────
PKG_CONFIG_PATH=$HOME/local/lib/pkgconfig gcc -O2 -Wall -Wextra -std=c11 maze_https_mongo.c -o maze_https_final $(pkg-config --cflags --libs libmicrohttpd gnutls libbson-1.0 libmongoc-1.0) -I$HOME/local/include -L$HOME/local/lib -lhiredis -Wl,-rpath,$HOME/local/lib

# ─────────────────────────────────────────
# STEP 2 — Confirm Redis is running
# ─────────────────────────────────────────
redis-cli ping
# Expected: PONG

# ─────────────────────────────────────────
# STEP 3 — Start the server (keep this terminal open)
# ─────────────────────────────────────────
REDIS_HOST=127.0.0.1 REDIS_PORT=6379 REDIS_KEY_PREFIX=team3ttmission ./maze_https_final

# ─────────────────────────────────────────
# STEP 4 — Health check (new terminal)
# ─────────────────────────────────────────
curl --cacert certs/ca.crt --cert certs/client.crt --key certs/client.key https://10.170.8.109:8445/health
# Expected: {"ok":true}

# ─────────────────────────────────────────
# STEP 5 — POST player telemetry → MongoDB
# ─────────────────────────────────────────
curl --cacert certs/ca.crt --cert certs/client.crt --key certs/client.key -X POST https://10.170.8.109:8445/move -H "Content-Type: application/json" -d '{"event_type":"player_move","input":{"device":"keyboard","move_sequence":1},"player":{"position":{"x":1,"y":0}},"goal_reached":false,"timestamp":"2026-02-17T03:30:00Z"}'
# Expected: {"ok":true}

# ─────────────────────────────────────────
# STEP 6 — GET telemetry back from MongoDB
# ─────────────────────────────────────────
curl --cacert certs/ca.crt --cert certs/client.crt --key certs/client.key "https://10.170.8.109:8445/moves?limit=1"
# Expected: {"ok":true,"moves":[{...}],"count":1,"truncated":false}

# ─────────────────────────────────────────
# STEP 7 — POST mission data → Redis
# ─────────────────────────────────────────
curl --cacert certs/ca.crt --cert certs/client.crt --key certs/client.key -X POST https://10.170.8.109:8445/mission -H "Content-Type: application/json" -d '{"mission_id":"DEMO_MISSION","robot_id":"ROBOT_01","mission_type":"patrol","start_time":1770056813,"end_time":1770056848,"moves_left_turn":46,"moves_right_turn":46,"moves_straight":52,"moves_reverse":8,"moves_total":152,"distance_traveled":24.41,"duration_seconds":35,"mission_result":"success","abort_reason":""}'
# Expected: {"ok":true,"key":"team3ttmission:DEMO_MISSION"}

# ─────────────────────────────────────────
# STEP 8 — GET mission back from Redis via HTTPS
# ─────────────────────────────────────────
curl --cacert certs/ca.crt --cert certs/client.crt --key certs/client.key "https://10.170.8.109:8445/mission?mission_id=DEMO_MISSION"
# Expected: {"ok":true,"key":"team3ttmission:DEMO_MISSION","mission":{...}}

# ─────────────────────────────────────────
# STEP 9 — Verify mission directly in Redis CLI
# ─────────────────────────────────────────
redis-cli HGETALL team3ttmission:DEMO_MISSION
# Expected: all mission fields listed as key-value pairs

# ─────────────────────────────────────────
# SECURITY TEST 1 — No client certificate → rejected
# ─────────────────────────────────────────
curl --cacert certs/ca.crt -X POST https://10.170.8.109:8445/move -H "Content-Type: application/json" -d '{"event_type":"player_move"}'
# Expected: {"error":"client certificate required"}  HTTP 403

# ─────────────────────────────────────────
# SECURITY TEST 2 — Rogue certificate not signed by MazeLab-CA → rejected
# ─────────────────────────────────────────
openssl req -x509 -newkey rsa:2048 -keyout /tmp/rogue.key -out /tmp/rogue.crt -days 1 -nodes -subj "/CN=rogue-client"

curl --cacert certs/ca.crt --cert /tmp/rogue.crt --key /tmp/rogue.key -X POST https://10.170.8.109:8445/move -H "Content-Type: application/json" -d '{"event_type":"player_move"}'
# Expected: {"error":"client certificate not signed by trusted CA"}  HTTP 403
